**Datei:** `BookingEntryForm.svelte:782-802, 804-840`

**Neue Funktion:**
```typescript
function selectDateSegment(index: number): void {
  const dateInput = document.getElementById('input-date') as HTMLInputElement;
  if (!dateInput || !dateInput.value) return;

  if (index === 0) dateInput.setSelectionRange(0, 2);      // MM
  else if (index === 1) dateInput.setSelectionRange(3, 5);  // DD
  else if (index === 2) dateInput.setSelectionRange(6, 10); // YYYY
}
```
